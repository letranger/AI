def get_slope(p1, p2):
    W = (p2.y - p1.y) / (p2.x - p1.x)
    return W

def get_bias(p1, W):
    B = p1.y - (W*p1.x)
    return B

def get_y(x, W, B):
    y = W*x + B

from dataclasses import dataclass    
@dataclass
class Point:
    x: float
    y: float

p2 = Point(3, 5)
